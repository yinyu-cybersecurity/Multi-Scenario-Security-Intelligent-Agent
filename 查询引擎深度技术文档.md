## 1. 查询引擎架构总览

### 1.1 核心设计理念

Claude Code 的查询引擎采用了**AsyncGenerator + 状态机**的设计:

```
用户输入 → QueryEngine.submitMessage() → query() → API 流式响应
                    ↓                          ↓
              状态管理                    工具执行
                    ↓                          ↓
              上下文压缩 ← ← ← ← ← ← ← ← ← 结果汇总
```

**关键设计原则**:

1. **AsyncGenerator 流式处理**: 所有消息通过 yield 流式返回
2. **状态机驱动**: 通过 Continue/Terminal 状态控制循环
3. **上下文隔离**: 每个查询有独立的上下文
4. **压缩感知**: 自动检测并压缩过长上下文
5. **错误恢复**: 支持 max_output_tokens 恢复和重试

### 2.2 QueryEngine 类结构

```typescript
export class QueryEngine {
  // ═══════════════════════════════════════════
  // 私有状态
  // ═══════════════════════════════════════════
  
  private config: QueryEngineConfig
  private mutableMessages: Message[]
  private abortController: AbortController
  private permissionDenials: SDKPermissionDenial[]
  private totalUsage: NonNullableUsage
  private hasHandledOrphanedPermission = false
  private readFileState: FileStateCache
  
  // 技能发现跟踪
  private discoveredSkillNames = new Set<string>()
  private loadedNestedMemoryPaths = new Set<string>()

  // ═══════════════════════════════════════════
  // 构造函数
  // ═══════════════════════════════════════════
  
  constructor(config: QueryEngineConfig) {
    this.config = config
    this.mutableMessages = config.initialMessages ?? []
    this.abortController = config.abortController ?? createAbortController()
    this.permissionDenials = []
    this.readFileState = config.readFileCache
    this.totalUsage = EMPTY_USAGE
  }

  // ═══════════════════════════════════════════
  // 核心方法
  // ═══════════════════════════════════════════
  
  async *submitMessage(
    prompt: string | ContentBlockParam[],
    options?: { uuid?: string; isMeta?: boolean },
  ): AsyncGenerator<SDKMessage, void, unknown> {
    // 1. 清理状态
    this.discoveredSkillNames.clear()
    setCwd(cwd)
    
    // 2. 包装 canUseTool 跟踪权限拒绝
    const wrappedCanUseTool = async (...) => {
      const result = await canUseTool(...)
      if (result.behavior !== 'allow') {
        this.permissionDenials.push({...})
      }
      return result
    }
    
    // 3. 获取初始状态
    const initialAppState = getAppState()
    const initialMainLoopModel = userSpecifiedModel
      ? parseUserSpecifiedModel(userSpecifiedModel)
      : getMainLoopModel()
    
    // 4. 构建系统提示
    const {
      defaultSystemPrompt,
      userContext,
      systemContext,
    } = await fetchSystemPromptParts({...})
    
    // 5. 处理用户输入
    const processedInput = await processUserInput(prompt, {...})
    
    // 6. 构建工具使用上下文
    const toolUseContext: ToolUseContext = {
      options: {...},
      abortController: this.abortController,
      readFileState: this.readFileState,
      messages: this.mutableMessages,
      // ...
    }
    
    // 7. 调用 query 循环
    for await (const event of query({
      messages: this.mutableMessages,
      systemPrompt,
      userContext,
      systemContext,
      canUseTool: wrappedCanUseTool,
      toolUseContext,
      // ...
    })) {
      // 8. yield SDK 消息
      yield normalizeMessage(event)
    }
  }

  // ═══════════════════════════════════════════
  // 辅助方法
  // ═══════════════════════════════════════════
  
  getMessages(): Message[] {
    return [...this.mutableMessages]
  }
  
  getUsage(): NonNullableUsage {
    return this.totalUsage
  }
  
  abort() {
    this.abortController.abort()
  }
}
```

```typescript
  params: QueryParams,
): AsyncGenerator<
  | StreamEvent           // 流事件 (增量更新)
```

### 3.3 queryLoop 核心循环

```typescript
async function* queryLoop(
  params: QueryParams,
  consumedCommandUuids: string[],
): AsyncGenerator<..., Terminal> {
  
  // ═══════════════════════════════════════════
  // 初始化不可变参数
  // ═══════════════════════════════════════════
  
  const {
    systemPrompt,
    userContext,
    systemContext,
    canUseTool,
    fallbackModel,
    querySource,
    maxTurns,
    skipCacheWrite,
  } = params
  const deps = params.deps ?? productionDeps()

  // ═══════════════════════════════════════════
  // 初始化可变状态
  // ═══════════════════════════════════════════
  
  let state: State = {
    messages: params.messages,
    toolUseContext: params.toolUseContext,
    maxOutputTokensOverride: params.maxOutputTokensOverride,
    autoCompactTracking: undefined,
    stopHookActive: undefined,
    maxOutputTokensRecoveryCount: 0,
    hasAttemptedReactiveCompact: false,
    turnCount: 1,
    pendingToolUseSummary: undefined,
    transition: undefined,
  }
  
  const budgetTracker = createBudgetTracker()
  let taskBudgetRemaining: number | undefined = undefined
  const config = buildQueryConfig()

  // ═══════════════════════════════════════════
  // 主循环
  // ═══════════════════════════════════════════
  
  while (true) {
    // 1. 解构当前状态
    let {
      messages,
      toolUseContext,
      autoCompactTracking,
      maxOutputTokensRecoveryCount,
      hasAttemptedReactiveCompact,
      maxOutputTokensOverride,
      pendingToolUseSummary,
      stopHookActive,
      turnCount,
      transition,
    } = state

    // 2. 检查中止
    if (toolUseContext.abortController.signal.aborted) {
      return { type: 'aborted' }
    }

    // 3. 检查最大轮次
    if (maxTurns !== undefined && turnCount > maxTurns) {
      return { type: 'max_turns', maxTurns }
    }

    // 4. 归一化消息
    const { messagesForAPI, pendingToolResults } = normalizeMessagesForAPI(messages, {...})

    // 5. 检查上下文大小
    const tokenCount = await tokenCountWithEstimation(messagesForAPI)
    if (tokenCount > MAX_CONTEXT_TOKENS) {
      // 触发压缩
      const compacted = await compact(messages, {...})
      state = { ...state, messages: compacted.messages }
      continue
    }

    // 6. 检查自动压缩
    if (isAutoCompactEnabled() && shouldAutoCompact(tokenCount)) {
      const autoCompacted = await autoCompact(messages, {...})
      state = { ...state, messages: autoCompacted.messages }
      continue
    }

    // 7. 调用 API
    const stream = await deps.claude({
      messages: messagesForAPI,
      systemPrompt,
      userContext,
      systemContext,
      tools: toolUseContext.options.tools,
      model: toolUseContext.options.mainLoopModel,
      maxOutputTokens: maxOutputTokensOverride,
      // ...
    })

    // 8. 处理流式响应
    let assistantMessage: AssistantMessage | undefined
    let hasToolUse = false
    
    for await (const event of stream) {
      if (event.type === 'stream_event') {
        // 增量更新
        yield event
      } else if (event.type === 'assistant') {
        assistantMessage = event
        messages.push(event)
        hasToolUse = event.message.content.some(b => b.type === 'tool_use')
      }
    }

    // 9. 检查 max_output_tokens 错误
    if (assistantMessage?.apiError === 'max_output_tokens') {
      maxOutputTokensRecoveryCount++
      if (maxOutputTokensRecoveryCount <= MAX_OUTPUT_TOKENS_RECOVERY_LIMIT) {
        // 恢复：继续生成
        state = {
          ...state,
          messages,
          maxOutputTokensRecoveryCount,
          transition: { type: 'max_output_tokens_recovery' },
        }
        continue
      }
      return { type: 'max_output_tokens_exhausted' }
    }

    // 10. 处理工具调用
    if (hasToolUse) {
      const toolResults = await runTools({
        assistantMessage,
        toolUseContext,
        canUseTool,
        messages,
      })
      
      messages.push(...toolResults.newMessages)
      
      state = {
        ...state,
        messages,
        turnCount: turnCount + 1,
        transition: { type: 'tool_use' },
      }
      continue
    }

    // 11. 无工具调用 - 完成
    return { type: 'complete' }
  }
}
```

## 4. 状态机设计

```typescript
type Terminal =
```

```typescript
export async function runTools(params: {
  assistantMessage: AssistantMessage
  toolUseContext: ToolUseContext
  canUseTool: CanUseToolFn
  messages: Message[]
}): Promise<{ newMessages: Message[] }> {
  const { assistantMessage, toolUseContext, canUseTool, messages } = params
  
  // 1. 提取所有 tool_use 块
  const toolUseBlocks = assistantMessage.message.content.filter(
    (block): block is ToolUseBlock => block.type === 'tool_use'
  )
  
  // 2. 分类工具 (并发安全 vs 非并发安全)
  const { safe, unsafe } = categorizeToolUses(toolUseBlocks, toolUseContext)
  
  // 3. 并行执行并发安全工具
  const safeResults = await Promise.all(
    safe.map(block => executeToolUse(block, toolUseContext, canUseTool))
  )
  
  // 4. 串行执行非并发安全工具
  const unsafeResults: ToolResult[] = []
  for (const block of unsafe) {
    const result = await executeToolUse(block, toolUseContext, canUseTool)
    unsafeResults.push(result)
  }
  
  // 5. 构建结果消息
  const newMessages = buildResultMessages([...safeResults, ...unsafeResults])
  
  return { newMessages }
}
```

### 7.2 executeToolUse 执行单个工具

```typescript
async function executeToolUse(
  toolUseBlock: ToolUseBlock,
  toolUseContext: ToolUseContext,
  canUseTool: CanUseToolFn
): Promise<ToolResult> {
  // 1. 查找工具
  const tool = findToolByName(toolUseContext.options.tools, toolUseBlock.name)
  if (!tool) {
    return createErrorResult(toolUseBlock, `Unknown tool: ${toolUseBlock.name}`)
  }
  
  // 2. 解析输入
  const input = tool.inputSchema.parse(toolUseBlock.input)
  
  // 3. 验证输入
  if (tool.validateInput) {
    const validation = await tool.validateInput(input, toolUseContext)
    if (validation.result === false) {
      return createErrorResult(toolUseBlock, validation.message)
    }
  }
  
  // 4. 检查权限
  const permission = await tool.checkPermissions(input, toolUseContext)
  if (permission.behavior === 'deny') {
    return createDeniedResult(toolUseBlock, permission.message)
  }
  
  // 5. 调用 canUseTool (可能触发用户确认)
  const canUseResult = await canUseTool(
    tool,
    input,
    toolUseContext,
    assistantMessage,
    toolUseBlock.id
  )
  
  if (canUseResult.behavior !== 'allow') {
    return createDeniedResult(toolUseBlock, canUseResult.message)
  }
  
  // 6. 执行工具
  try {
    const result = await tool.call(
      canUseResult.updatedInput ?? input,
      toolUseContext,
      canUseTool,
      assistantMessage,
      (progress) => {
        // 进度回调
        emitProgress(progress)
      }
    )
    
    return result
  } catch (error) {
    return createErrorResult(toolUseBlock, error.message)
  }
}
```

```typescript
function categorizeRetryableAPIError(error: APIError): {
  retryable: boolean
  retryDelay?: number
} {
  switch (error.type) {
    case 'overloaded':
      return { retryable: true, retryDelay: 1000 }
    case 'rate_limit':
      return { retryable: true, retryDelay: error.retryAfter ?? 5000 }
    case 'timeout':
      return { retryable: true, retryDelay: 0 }
    default:
      return { retryable: false }
  }
}

// 带重试的 API 调用
async function callAPIWithRetry(params: APIParams, maxRetries = 3): Promise<APIResponse> {
  let lastError: Error | undefined
  
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      return await callAPI(params)
    } catch (error) {
      const categorized = categorizeRetryableAPIError(error)
      
      if (!categorized.retryable) {
        throw error
      }
      
      lastError = error
      
      if (categorized.retryDelay) {
        await sleep(categorized.retryDelay)
      }
    }
  }
  
  throw lastError
}
```
